# 2번
class Student:
    def __init__(self, name, student_id):
        self.name = name
        self.student_id = student_id
        self.grades = {}
    def add_grade(self, subject, score):
        self.grades[subject] = score
    def get_average_grade(self):
        avg = 0.0
        for i in self.grades:
            avg += self.grades[i]
        avg /= len(self.grades)
        return round(avg,1)

s1 = Student('김일','001')

s1.add_grade('국어',100)
s1.add_grade('수학',90)
s1.add_grade('영어',80)

print(f'학생 이름 : {s1.name}')
print(f'평균 성적 {s1.get_average_grade()}')

anynumber = int(input('몇 개의 숫자를 입력하시겠습니까? >>> '))
numbers = []
plus = 0
minus = 0
zero = 0
for i in range(anynumber):
    numbers.append(int(input(f'{i+1}번째 숫자를 입력하시오 >>>')))
    if numbers[i] > 0:
        plus += 1
    elif numbers[i] < 0:
        minus += 1
    else :
        zero += 1

print(f'양수: {plus}개')
print(f'음수: {minus}개')
print(f'0: {zero}개')

# 4번
anynumber = int(input('후보자 수를 입력하시오 >>> '))
candidates = []
for i in range(anynumber):
    candidates.append(input(f'{i+1}번째 후보자의 이름을 입력하시오 >>> '))

votes = {}
for candidate in candidates:
    votes[candidate] = 0

info = ''
for symbolN in range(len(candidates)):
    if symbolN == len(candidates)-1:
        info += str(symbolN+1)+': '+candidates[symbolN]
    else :
        info += str(symbolN+1)+': '+candidates[symbolN]+', '


anynumber2 = int(input('전체 투표 횟수를 입력하시오 >>> '))
for j in range(anynumber2):
    voteN = int(input(f'{j+1}번째 투표 ({info}) >>> '))
    if voteN > len(candidates):
        print('해당 후보는 없습니다.')
    else :
        votes[candidates[voteN-1]] += 1

print('--- 투표 결과 ---')

for cd in votes:
    print(f'{cd} : {votes[cd]}표')
